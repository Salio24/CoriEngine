117 void Window::OnUpdate() {
118 CORI_PROFILE_FUNCTION();
119 SDL_Event e;
120
121 while (SDL_PollEvent(&e)) {
122 ImGui_ImplSDL3_ProcessEvent(&e);
123
124 // ReSharper disable once CppDefaultCaseNotHandledInSwitchStatement
125 switch (e.type) {
126 case SDL_EVENT_WINDOW_RESIZED:
127 {
128 if (e.window.windowID != SDL_GetWindowID(m_Data->m_Window)) {
129 break;
130 }
131
132 if (m_Data->m_CurrentWindowMode != WindowMode::BORDERLESS_WINDOWED) {
133 WindowResizeEvent resizeEvent(m_Data->m_CurrentScreenMode.m_Width, m_Data->m_CurrentScreenMode.m_Height);
134 m_Data->m_EventCallback(resizeEvent);
135 } else {
136 WindowResizeEvent resizeEvent(m_Data->m_SDLModes[0]->w, m_Data->m_SDLModes[0]->h);
137 m_Data->m_EventCallback(resizeEvent);
138 }
139
140 break;
141 }
142 case SDL_EVENT_WINDOW_CLOSE_REQUESTED:
143 {
144 WindowCloseEvent closeEvent;
145 m_Data->m_EventCallback(closeEvent);
146 break;
147 }
148 case SDL_EVENT_KEY_DOWN:
149 {
150 KeyPressedEvent keyPressedEvent(static_cast<CoriKeycode>(e.key.scancode), e.key.repeat);
151 m_Data->m_EventCallback(keyPressedEvent);
152 break;
153 }
154 case SDL_EVENT_KEY_UP:
155 {
156 KeyReleasedEvent keyReleasedEvent(static_cast<CoriKeycode>(e.key.scancode));
157 m_Data->m_EventCallback(keyReleasedEvent);
158 break;
159 }
160 case SDL_EVENT_MOUSE_MOTION:
161 {
162 MouseMovedEvent mouseMovedEvent(static_cast<int32_t>(e.motion.x), static_cast<int32_t>(e.motion.y));
163 m_Data->m_EventCallback(mouseMovedEvent);
164 break;
165 }
166 case SDL_EVENT_MOUSE_WHEEL:
167 {
168 MouseScrolledEvent mouseScrolledEvent(static_cast<int16_t>(e.wheel.x), static_cast<int16_t>(e.wheel.y));
169 m_Data->m_EventCallback(mouseScrolledEvent);
170 break;
171 }
172 case SDL_EVENT_MOUSE_BUTTON_DOWN:
173 {
174 MouseButtonPressedEvent mouseButtonPressedEvent(static_cast<CoriMouseCode>(e.button.button));
175 m_Data->m_EventCallback(mouseButtonPressedEvent);
176 break;
177 }
178 case SDL_EVENT_MOUSE_BUTTON_UP:
179 {
180 MouseButtonReleasedEvent mouseButtonReleasedEvent(static_cast<CoriMouseCode>(e.button.button));
181 m_Data->m_EventCallback(mouseButtonReleasedEvent);
182 break;
183 }
184 }
