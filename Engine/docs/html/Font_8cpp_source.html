20 coriFont.reset(new Font(static_cast<void*>(fontPlaceholder), charsets, path, minimalScale, miterLimit));
21 CORI_CORE_ERROR_TAGGED({ Logger::Tags::Graphics::Self, Logger::Tags::Graphics::Font }, "Failed to load Font from '{}', loaded bundled placeholder font instead.", path.string());
22 msdfgen::destroyFont(fontPlaceholder);
